Enum sexo_enum {
  masculino
  feminino
  outro
}

Enum tipo_atividade_enum {
  artigo
  comunicacao
  congresso
  poster
  outro
}

Enum tipo_formacao_enum {
  curso
  congresso
  pos_graduacao
  mestrado
  doutoramento
  outro
}

Enum admin_role_enum {
  super_admin
  admin
}


// =====================================================
// USERS
// =====================================================

Table users {
  id bigint [pk]
  name varchar
  email varchar [unique]
  is_active boolean [default: true]

  hospital_id bigint [null, ref: > hospitals.id]
  especialidade_id bigint [null, ref: > especialidades.id]

  email_verified_at timestamp [null]
  password varchar
  two_factor_secret text [null]
  two_factor_recovery_codes text [null]
  two_factor_confirmed_at timestamp [null]
}


// =====================================================
// UTENTES
// =====================================================

Table utentes {
  id bigint [pk]
  nome varchar [null]
  idade int
  sexo sexo_enum
  processo int
  user_id bigint [null, ref: > users.id]
}


// =====================================================
// REGISTO CIRURGICOS
// =====================================================

Table registo_cirurgicos {
  id bigint [pk]

  user_id bigint [null, ref: > users.id]
  utente_id bigint [not null, ref: > utentes.id]

  hospital_id bigint [null, ref: > hospitals.id]
  especialidade_id bigint [null, ref: > especialidades.id]

  data_cirurgia date

  tipo_de_cirurgia_id bigint [
    not null,
    ref: > tipo_de_cirurgias.id
  ]

  tipo_de_abordagem_id bigint [
    null,
    ref: > tipo_de_abordagens.id
  ]

  ambulatorio boolean [default: false]
  observacoes text [null]
}

  ]

  procedimento_id bigint [
    not null,

  ]

    null,

  ]

  clavien_dindo varchar [

    note: 'Coluna clavien-dindo'
  ]

  observacoes text [null]
}

// =====================================================
// ROLES
// =====================================================

Table roles {
  id bigint [pk]

  name varchar
  guard_name varchar

  Indexes {
    (name, guard_name) [unique]
  }
}


// =====================================================
// PERMISSIONS
// =====================================================

Table permissions {
  id bigint [pk]

  name varchar
  guard_name varchar

  Indexes {
    (name, guard_name) [unique]
  }
}


// =====================================================
// MODEL HAS ROLES
// Laravel polymorphic relationship
// =====================================================

Table model_has_roles {
  role_id bigint [
    not null,
    ref: > roles.id
  ]

  model_type varchar [not null]
  model_id bigint [not null]

  Indexes {
    (role_id, model_id, model_type) [pk]
    model_id
    model_type
  }
}


// =====================================================
// ROLE HAS PERMISSIONS
// =====================================================

Table role_has_permissions {
  permission_id bigint [
    not null,
    ref: > permissions.id
  ]

  role_id bigint [
    not null,
    ref: > roles.id
  ]

  Indexes {
    (permission_id, role_id) [pk]
  }
}
